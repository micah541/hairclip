\section{The theorem and the hand-off}\label{M:sec:thm}
\begin{theorem}[trapping near $S_1$ with a floor, v3.3]\label{M:thm:W}
Let $\gamma$ be a smooth closed immersed curve evolving by CDF and $t_1$ a time at which the following hold.
\begin{enumerate}[label=\textup{(H\arabic*)},leftmargin=*]
\item \emph{Modulation and smallness}: the modulated decomposition exists at $\tau_1$, $|\alpha(\tau_1)|\le\alpha_1=1.5\cdot10^{-4}$, $\sqrt{\Lambda(\tau_1)}\le5\cdot10^{-8}$.
\item \emph{Shell energies, tightened}: on $y\in[14,60]$, $E_j(\tau_1)\le\HtwoV$ ($\omega$-normalised, $\omega=28/y$).
The variable is $f-U_1$ (B0 route (b)); certified by the step-4 stage \fn{h2u_v33}: $\HtwoCertV$, margins $\HtwoMarV$ \tC\ (Remark~\ref{A:rem:B0}, (I4)).
\item[\textup{(H2f)}] \emph{Far datum of (A)}: $G_j=\int_{\ln60}^{\ln700}\frac{28}y\bigl(\partial_s^j(f-U_1)\bigr)^2ds\le\HtwofV$, $j=0,1,2$.
\item[\textup{(H3)}] \emph{Windows}: on every $[y_w,1.5y_w]$, $y_w\in[50,253.125]$, the energies of $v=f-U_1$ satisfy $E_0\le2\cdot10^{-7}$, $E_1,E_2,E_3\le10^{-5}$, and $K_4\le0.06$ on the cone cylinders at $\tau_1$ (as v3.2; Lemma~U uses levels $0$--$2$).
\item[\textup{(H4)}] \emph{Graph}: the arc is a graph over the $S_1$ ray on $y\in[48,700]$ (v3.2: $[48,350]$; extended for (H2f) and Lemma~G, \S\ref{C:sec:graph}). \item[\textup{(H5)}] \emph{Lemma~O data}: hypotheses H1--H7, H9 of Lemma~O for the fixed-$L$ cone family ($L=50\lambda_1$, $X=3$, centres $[300,380]\lambda_1$, $(M,K)=(0.12,0.18)$: slope $\le0.08$, $|h_0''|\le0.03$, $\|h_0'''\|_{L^2}\le0.004$, jets $\le0.12$, $\int(h_0^{(5)})^2\le0.0144$, graph test, time ratio) and for the far family (centres $\ge490\lambda_1$, $(M,K)=(0.15,0.3)$, data as v3.2), arc by arc.
\item[\textup{(H6)}] \emph{Regularity and symmetry}: $\gamma(t_1)$ is a smooth closed immersed curve, \emph{symmetric about the $x$-axis} (text v40, review fix18 F-3: the symmetry is used by U-1, through (S4.2) and its Lemma ``symmetry'', and by the even class of the verifier); the forcing norm is determined by $\alpha(\tau_1)$. (For the clip, (H6) is Theorem C-1 (i), (iv) of \fn{C1_identification_v41.tex}, not step~4's [A] sentence; the symmetry comes from the symmetric $\Gamma_0$ of step~4 and uniqueness; step-4 wording, which states the symmetry: \fn{step4_handoff_v41.tex} item (3).)
\item[\textup{(GS)}] \emph{Tip chart}: on $|x|\le60$ the tip arc is a normal graph $u$ over $S_1$ with $\sup|ku|\le\frac12$, $\sup|u'|\le1$ (certified for the clip on $|x|\le60\frac14$ with $6.52\cdot10^{-3}$, $0.0542$, \fn{fix14_interface/GS_tau1.json}, pinned as stage 23 of \fn{FINAL_manifest_v38.json} (29 stages, \fn{be909bbf...}); Lemma GS-1 of \fn{GS_lemma_v41.tex}; hand-off item (4) of \fn{step4_handoff_v41.tex}).
\end{enumerate}
Then the solution exists on $[t_1,T)$ with $T<\infty$, the modulated decomposition persists, and (W1)--(W5) of \S\ref{C:sec:W} hold.
\end{theorem}
