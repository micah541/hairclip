\section{Statement}\label{A:sec:statement}
\begin{proposition}[level-$j$ inequality]\label{A:prop:level}
Assume on $[\tau_1,\tau']$: the verifier conclusions D1 (B1, $\bar a,\bar b,\bar\beta$, the pointwise ramp values, H-J-W of Lemma~\ref{A:lem:HJ}); the boxes of \S\ref{A:sec:boxes}; H-O2$'$, $\sup_\tau\int_O(z^A_k)^2ds\le\Nb_k^2$ ($k=0,1,2$); and the profile bound $|\alpha|\le\alpha_1=1.5\cdot10^{-4}$. Then for $j=0,1,2$ and a.e.\ $\tau$,
\begin{equation}\label{A:eq:D1}
\frac{d}{d\tau}E^A_j+\gamma'_jE^A_j+r_j\,a_c\!\int_{\ln28}^{\ln350}\!\varepsilon\omega\,(z^A_{j+2})^2ds\ \le\ \mathfrak s_j(\tau)+2\sqrt{E^A_j}\,F_j+\mathrm{coup}_j(\tau)+\oh_j ,
\end{equation}
where
\begin{itemize}[leftmargin=*]
\item $\gamma'_j=c_{\min}-(|1-2j|+1)\bar b/14-28^{-4}\hat G^{\rm core}_{jj}\ge\gpV$ \tC\ (\SA\ \key{gp});
\item $r_0=r_1=\rexpV$, $r_2=0$, $a_c=(1+0.11^2)^{-2}\ge\acV$ \tC\ (\SA\ \key{r_export}, \key{amin_c});
\item $\mathfrak s_j\ge0$ is a sum of densities supported on the inner ramp $[14,28]$: inflow, jet-free cutoff pairs and the ramp share of the rate, top pair, cutoff$\times$jet classes, foot and S33 terms (\S\ref{A:sec:ramp}--\ref{A:sec:foot}); each is a quadratic, cubic or product expression in the $z^A_k$ on the ramp;
\item $F_j^2=\int\omega|\partial_s^jR|^2\le(r_0^Fp^j)^2I_{\rm int}$ (\S\ref{A:sec:forcing});
\item $\mathrm{coup}_j(\tau)=\sum_{k<j}\hat G^{\rm core}_{jk}28^{-4}E^A_k(\tau)$, with $\GcoreOffV$ \tC\ (\SA\ \key{interface28.Ghat_core});
\item $\oh_j=\sum_{k=0}^2\kappa^o_{jk}\Nb_k^2$, $\kappa^o$ \tC\ (\SA\ \key{kappa_o}; \fn{outer_L2.py}).
\end{itemize}
\end{proposition}

\begin{theorem}[repaired Lemma S(A)]\label{A:thm:A}
Assume Proposition~\ref{A:prop:level}, the datum $E^A_j(\tau_1)\le H_{2,j}+H^f_{2,j}$, and the outer data $\Nb^2$ of Lemma~N at $\beta=2$. Then on $[\tau_1,\tau']$:

\begin{lemma}[integration of \eqref{A:eq:D1}]\label{A:lem:int}
Let $D^{\rm src}_j(\gamma'')$ denote the value of the ramp data computed with $J^e(\gamma'')$ of Corollary~\ref{A:cor:exp} (all densities of $\mathfrak s_j$ converted by Lemma~\ref{A:lem:conv}, Cauchy--Schwarz and H\"older applied with the measure $e^{-\gamma''(\tau-\sigma)}d\sigma\,ds$, and $\Phi$ of Lemma~\ref{A:lem:Phi} with the same $\gamma''$), with the $d$-parts $t_d$ (rates per unit $\tau$) contributing $t_d/\gamma''$ after the $\eta$-split. \begin{align*}
\text{(R-half)}\quad &E^A_j(\tau)\le E^A_j(\tau_1)+D^{\rm src}_j(\tfrac12\gamma'_j)+\frac{2(\hat c_j+\oh_j)}{\gamma'_j}+\frac{4F_j^2}{\gamma_j'^2},\\
\text{(R-sup)}\quad &E^A_j(\tau)\le A_j+\frac{2F_j^2}{\gamma_j'^2}+\frac{2F_j}{\gamma'_j}\Bigl(A_j+\frac{F_j^2}{\gamma_j'^2}\Bigr)^{1/2},\qquad A_j:=E^A_j(\tau_1)+D^{\rm src}_j(\gamma'_j)+\frac{\hat c_j+\oh_j}{\gamma'_j}.
\end{align*}
$\Eh_j$ is the smaller of the two right-hand sides.
\end{lemma}
\begin{proof}\tP\ (R-half): $2\sqrt EF\le\frac{\gamma'}2E+2F^2/\gamma'$, so $E'+\frac{\gamma'}2E\le\mathfrak s+\hat c+\oh+2F^2/\gamma'$; variation of constants with rate $\gamma'/2$ and Corollary~\ref{A:cor:exp} with $\gamma''=\gamma'/2$. The data terms are nonnegative, so the cubic and product classes are bounded by H\"older/Cauchy--Schwarz in the weighted measure exactly as for $\gamma''$ in the scripts.
\end{proof}
