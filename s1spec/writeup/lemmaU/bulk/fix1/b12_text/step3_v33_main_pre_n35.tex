\section{The theorem and the hand-off}\label{M:sec:thm}
\begin{theorem}[trapping near $S_1$ with a floor, v3.3]\label{M:thm:W}
Let $\gamma$ be a smooth closed immersed curve evolving by CDF and $t_1$ a time at which the following hold.
\begin{enumerate}[label=\textup{(H\arabic*)},leftmargin=*]
\item \emph{Modulation and smallness}: the modulated decomposition exists at $\tau_1$, $|\alpha(\tau_1)|\le\alpha_1=1.5\cdot10^{-4}$, $\sqrt{\Lambda(\tau_1)}\le5\cdot10^{-8}$.
\item \emph{Shell energies, tightened}: on $y\in[14,60]$, $E_j(\tau_1)\le\HtwoV$ ($\omega$-normalised, $\omega=28/y$).
The variable is $f-U_1$ (B0 route (b)); certified by the step-4 stage \fn{h2u_v33}: $\HtwoCertV$, margins $\HtwoMarV$ \tC\ (Remark~\ref{A:rem:B0}, (I4)).
\item[\textup{(H2f)}] \emph{Far datum of (A)}: $G_j=\int_{\ln60}^{\ln700}\frac{28}y\bigl(\partial_s^j(f-U_1)\bigr)^2ds\le\HtwofV$, $j=0,1,2$.
\item[\textup{(H3)}] \emph{Windows}: on every $[y_w,1.5y_w]$, $y_w\in[50,253.125]$, the energies of $v=f-U_1$ satisfy $E_0\le2\cdot10^{-7}$, $E_1,E_2,E_3\le10^{-5}$, and $K_4\le0.06$ on the cone cylinders at $\tau_1$ (as v3.2; Lemma~U uses levels $0$--$2$).
\item[\textup{(H4)}] \emph{Graph}: the arc is a graph over the $S_1$ ray on $y\in[48,700]$ (v3.2: $[48,350]$; extended for (H2f) and Lemma~G, \S\ref{C:sec:graph}; proposed wording: the step-4 (H2f) stage evaluates $f$ on $x\in[45,725]$, which implies it, but the hand-off list must say so).
\item[\textup{(H5)}] \emph{Lemma~O data}: hypotheses H1--H7, H9 of Lemma~O for the fixed-$L$ cone family ($L=50\lambda_1$, $X=3$, centres $[300,380]\lambda_1$, $(M,K)=(0.12,0.18)$: slope $\le0.08$, $|h_0''|\le0.03$, $\|h_0'''\|_{L^2}\le0.004$, jets $\le0.12$, $\int(h_0^{(5)})^2\le0.0144$, graph test, time ratio) and for the far family (centres $\ge490\lambda_1$, $(M,K)=(0.15,0.3)$, data as v3.2), arc by arc.
\item[\textup{(H6)}] \emph{Regularity}: $\gamma(t_1)$ smooth; the forcing norm is determined by $\alpha(\tau_1)$.
\end{enumerate}
Then the solution exists on $[t_1,T)$ with $T<\infty$, the modulated decomposition persists, and (W1)--(W5) of \S\ref{C:sec:W} hold.
\end{theorem}
