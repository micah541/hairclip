\begin{remark}[endpoint terms; why not the trace form]\label{F:rem:trace}
(1) At level 0 the only integration by parts has a coefficient that vanishes at both ends. At level 2 there is none. In particular the point term $W(57)\frac{280}{27}g(57)g'''(57)$ of the first flux lemma (F1) does not arise: the monomial $W\zeta\zeta''''gg''''$ is bounded by $\frac{280}{27}N_0N_4$.
(2) The route ``integrate $g''''$ away and bound the trace $g'''(57)$ in $L^2(d\tau)$'' does not close $K_2^{\OR}$: $\zeta N_{\rm low}'$ contains $d_{110}(g'g'')'$, and $(g'g'')'g''''+\frac52g''g'''^2$ is a total derivative \tS, so one is left with $-\frac52\int W\zeta^2d_{110}\,g''g'''^2$, which is not integrable in $\tau$ when $g'''$ is controlled only in $L^2(d\tau;L^2_x)$ and $g''$ has no pointwise-in-$\tau$ sup bound.
\end{remark}

\subsection{Absorption of the time-integrated part}\label{F:sec:abs}
Let $\mathrm{tot}_s(\alpha,x)$ be the verifier's right-hand side without the flux constants and $\mathrm{tot}:=\mathrm{tot}_s+F_0$ the function its certificates bound. The $E_I$ inequality and Lemma~\ref{F:lem:flux}(d) give, for a.e.\ $\tau$,
\begin{equation}\label{F:eq:star}
\tfrac12\Lambda'\le\mathrm{tot}(\alpha(\tau),x(\tau))-s+\varphi(\tau).
\end{equation}

\begin{lemma}[reflected integral]\label{F:lem:Z}
Let $\Lambda$ be absolutely continuous on $[\tau_1,\tau']$ and satisfy \eqref{F:eq:star}; let $\varphi\ge0$ be integrable with $\int_\sigma^\tau\varphi\le C'+S'(\tau-\sigma)$ for all $\tau_1\le\sigma\le\tau\le\tau'$, and $S'\le s$. Put $P(\tau)=2\int_{\tau_1}^\tau(\varphi-s)$ and
\[Z(\tau):=P(\tau)-\min_{[\tau_1,\tau]}P=2\sup_{\sigma\le\tau}\int_\sigma^\tau(\varphi-s).\]
Then $0\le Z\le2C'$, $Z(\tau_1)=0$, $Z$ is absolutely continuous with $Z'\ge2(\varphi-s)$ a.e., and $Y:=\Lambda-Z$ satisfies
\[\tfrac12Y'\le\mathrm{tot}(\alpha(\tau),x(\tau))\ \text{a.e.},\qquad Y\le\Lambda\le Y+2C' .\]
\end{lemma}
\begin{proof}\tHd\ $\int_\sigma^\tau(\varphi-s)\le C'+(S'-s)(\tau-\sigma)\le C'$, and $\sigma=\tau$ gives $Z\ge0$. $m(\tau)=\min_{[\tau_1,\tau]}P$ is nonincreasing and $|m(t)-m(r)|\le\int_r^t|P'|$, so $m$ is absolutely continuous with $m'\le0$ a.e.; hence $Z'=P'-m'\ge P'$. Subtract from \eqref{F:eq:star}.
\end{proof}

\begin{proposition}[the verifier's conclusions with $\Lambda\le\text{barrier}+2C'$]\label{F:prop:abs}
Let $(U_k,L_k,\mathrm{Ly}^{hi}_k)$ be the march cells of the rate table, $\mathrm{pad}_k=10^{-9}\max(1,|U_k|)$, and $(a^*,L^*,\rho,\eta,L_F)$ the tail data, for each of the two production points. Assume
\begin{enumerate}[label=(C\arabic*),leftmargin=3em,start=0]
\item $s>0$;\quad \textup{(C1)} $S'\le s$;
\setcounter{enumi}{1}
\item $L(\tau)+2C'\le\mathrm{Ly}^{hi}_k$ on every march cell;
\item $U_k^+\cdot2C'<\mathrm{pad}_k\,L_k$ for every $k$;
\item $2C'\le\eta L_F$;
\item the whole-tail $\alpha$-condition holds with its right side multiplied by $1+2C'/L_F$;
\item $\kappa_J^{\rm used}-\kappa_J^{\rm cert}\ge C'/X_{\rm const}$ (the unconditional integrals);
\item every other consumer of ``$\Lambda\le$ barrier'' uses a value with relative reserve $\ge2C'/L_k$ (march) or $\ge C'/L_F$ (floor).
\end{enumerate}
Then all conclusions of the verifier \fn{fin13v4b} hold on $[\tau_1,\tau']$ with $\Lambda(\tau)\le L(\tau)+2C'$ on the march and $\Lambda(\tau)\le\ell(\tau)+2C'$ in the tail, the $\alpha$-barriers unchanged, and the window integrals of $V_m^2$ bounded by $J_m(1+\kappa_J^{\rm used})+s_mT$.
\end{proposition}
\begin{proof}\tHd\ Fix one of the two production points and let $Y=\Lambda-Z$ be as in Lemma~\ref{F:lem:Z}; by (C0), (C1) the lemma applies, so $Y\le\Lambda\le Y+2C'$ and $\frac12Y'\le\mathrm{tot}(\alpha(\tau),x(\tau))$ a.e. Throughout, ``the state is in cell $k$'' means $\Lambda\in[\mathrm{Ly}^{lo}_k,\mathrm{Ly}^{hi}_k]$ and $|\alpha|\le\alpha^{\rm up}_k$; the alpha-barrier itself is checked by the verifier at $\mathrm{Ly}^{hi}_k$, and $\beta$-bounds are evaluated there.

\emph{What a cell certifies} \tR. \fn{closure.py}:129 calls \fn{orc.rate}, i.e.\ \fn{bb.bound_cell}$(0,\alpha_k,\mathrm{Ly}^{lo}_k,\mathrm{Ly}^{hi}_k,\text{floors}=\text{True})$, which returns $U_k\ge\sup\mathrm{tot}/\Lambda$ over all shapes, $|\alpha|\le\alpha_k$ and $\Lambda\in[\mathrm{Ly}^{lo}_k,\mathrm{Ly}^{hi}_k]$ (\fn{bb.py}:142--185; the $\Lambda$-range by evaluating $H-\mathrm{NEG}$ at $s_{hi}$ and $L$ at $s_{lo}$; the flux constants enter $L$ as the degree-0 constant $F_0$, taken at $\mathrm{Ly}^{lo}$; \fn{TAIL_LEMMA.tex}, ``s-monotone split'' and ``leaf certificate''). Cell ends: $\mathrm{Ly}^{hi}_k=\max(L_k,g)(1+10^{-3})$, $\mathrm{Ly}^{lo}_k=\min(L_k,g)(1-10^{-3})$, and the step is accepted only if $L_{k+1}\in[\mathrm{Ly}^{lo}_k,\mathrm{Ly}^{hi}_k]$ (\fn{closure.py}:112--115, 131--132). The barrier on step $k$ is $L'=2(U_k+\mathrm{pad}_k)L$, $\mathrm{pad}_k=10^{-9}\max(1,|U_k|)$ (\fn{closure.py}:130).

\emph{March (first touch with $Y$).} Claim: $Y\le L$ on every step. At $\tau_1$, $Z=0$ and $Y=\Lambda(\tau_1)\le\mathrm{Ly}_1<L(0)$. Suppose the claim fails on step $k$ and put $G:=Y-L$, absolutely continuous. Let $t_0$ be the last time before a point with $G>0$ at which $G\le0$; then $G(t_0)=0$ and $G>0$ on $(t_0,t_0+\delta)$. For $t\in(t_0,t_0+\delta)$ and $\delta$ small: $\Lambda\ge Y>L\ge\mathrm{Ly}^{lo}_k$, and $\Lambda\le Y+2C'=L+G+2C'<\mathrm{Ly}^{hi}_k$ because $L+2C'\le\mathrm{Ly}^{hi}_k$ holds with room on every step by (C2) (checked against $\max(L_k,L_{k+1})$, margin $\ge\times5.0\cdot10^{7}$) and $G$ is small. The $\alpha$-barrier is unchanged (it is evaluated at $\mathrm{Ly}^{hi}_k$ and $\Lambda\le\mathrm{Ly}^{hi}_k$). So the state is in cell $k$ and $\frac12Y'\le\mathrm{tot}\le U_k\Lambda$ a.e. on $(t_0,t_0+\delta)$.
If $U_k\ge0$: $\Lambda\le Y+2C'$ gives $Y'\le2U_k(Y+2C')$, and $L'=2(U_k+\mathrm{pad}_k)L$, so $G'\le2U_kG+(4U_kC'-2\,\mathrm{pad}_kL)$; the bracket is $<0$ by (C3) (with $L\ge\mathrm{Ly}^{lo}_k$, margin $\ge\times1547$). No march step of either table used the slack path of \fn{closure.py}:126--128 (\fn{recompute.json}, $n_{\rm slack}=0$).
