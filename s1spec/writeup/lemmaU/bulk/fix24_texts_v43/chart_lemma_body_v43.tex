\begin{lemma}[exact identification]\label{C:lem:id}
Where the curve is a graph over the ray ($Y'>0$ below), with $Y(\sigma):=y$ at the curve point $\Gamma(\sigma)$:
\begin{enumerate}
\item[(a)] $\operatorname{Re}(e^{-i\theta_\infty}\Gamma_1)=q\cos\delta-p\sin\delta$ and $\operatorname{Im}(e^{-i\theta_\infty}\Gamma_1)=q\sin\delta+p\cos\delta$; hence
\[Y=q\cos\delta-(p+u)\sin\delta,\qquad e:=Y-\sigma=(q-\sigma)\cos\delta-2\sigma\sin^2\tfrac\delta2-(p+u)\sin\delta,\qquad Y'=1-\eta,\]
\[\eta:=2\sin^2\tfrac\delta2+b\sin\delta+ku\cos\delta .\]
\item[(b)] $f(Y(\sigma))=b+\Phi$, \qquad $\Phi=\dfrac{(1-ku+b^2)\sin\delta+kub\cos\delta}{(1-ku)\cos\delta-b\sin\delta}$.
\item[(c)] $D_y=(1+\varepsilon_{\rm c})D$ along the curve, with $1+\varepsilon_{\rm c}=\dfrac{Y}{\sigma Y'}=\dfrac{1+e/\sigma}{1-\eta}$.
\item[(d)] For $k=0,\dots,3$:
\[
z^A_k\circ Y=D^k(w'+d)+\chi_k,\qquad \chi_k:=\bigl(T^k-D^k\bigr)b+T^k\Phi+p^k\bigl(U(\sigma)-U(Y)\bigr),\quad T=(1+\varepsilon_{\rm c})D .
\]
\item[(e)] $\chi_k=A_k+B_{k2}\,u'''+B_{k3}\,u''''$ with $A_k,B_{k2},B_{k3}$ explicit functions of $(\sigma,e,\delta,k,k',k'',k''',u,u',u'')$; $B_{02}=B_{03}=B_{12}=B_{13}=B_{23}=0$; $B_{22}$ does not depend on $u''$; every term of $A_k,B_{kj}$ carries a factor $\sin\delta$, $k^{(i)}$, $\eta$ or $e$, and $\chi_k\equiv0$ when $\delta\equiv0$, $e\equiv0$.
\end{enumerate}
\end{lemma}

\begin{proof}\tP\ (a) $e^{-i\theta_\infty}\Gamma_1=(q+ip)e^{i\delta}$. With $d(\sin\delta)/d\sigma=k\cos\delta$ and the profile ODE, $\frac{d}{d\sigma}(q\cos\delta-p\sin\delta)=(1+kp)\cos\delta-qk\sin\delta+kq\sin\delta-pk\cos\delta=\cos\delta$, so $Y'=\cos\delta-(u\sin\delta)'=1-\eta$. (b) $\Gamma'=T_1\bigl((1-ku)+i\,b\bigr)$, so the tangent angle of $\Gamma$ is $\theta_1+\arg((1-ku)+ib)$ and $f=\tan(\delta+\arctan\frac b{1-ku})$; expand $\tan(A+B)-b$. (c) $D_y=y\frac d{dy}=\frac{Y}{Y'}\frac d{d\sigma}$. (d) $z^A_k\circ Y=T^k(f\circ Y-U\circ Y)$; $T^k(U\circ Y)=p^kU(Y)$ since $D_y^ky^p=p^ky^p$; $f\circ Y=b+\Phi$ and $D^k(w'+d)=D^k(b-U(\sigma))=D^kb-p^kU(\sigma)$. (e) Symbolic: \fn{n1_chart.py} builds $\chi_k$ with sympy from (a)--(d) ($e'=-\eta$), asserts $\partial^2\chi_k/\partial(u''')^2=\partial^2\chi_k/\partial u'''\partial u''''=\partial^2\chi_k/\partial(u'''')^2=0$, no $u^{(5)}$, $\partial B_{22}/\partial u''=0$, and $\chi_k|_{\delta\equiv0,e=0}=0$.
\end{proof}
