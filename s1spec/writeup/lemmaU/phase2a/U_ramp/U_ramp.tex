\section{Transport identity for the moving weight}\label{sec:T}
\begin{lemma}[transport identity]\label{lem:T}
Let $z$ be smooth with $z(\cdot,\tau)$ compactly supported where $\Psi_\rho>0$ (P1 regularity), and let $\partial_\tau z_j=-c\,z_{j+1}+\partial_s^j(\beta z_1)+\mathcal N_j$, where $\mathcal N_j$ collects the fourth-order and forcing terms.
\begin{enumerate}[label=(\roman*),leftmargin=*]
\item \emph{(co-moving cancellation; exact)}
\[
\int\partial_\tau\Psi_\rho\,z_j^2+2\int\Psi_\rho z_j(-c\,z_{j+1})
=\ c\!\int(\eta^2)'\chi^2WD\phi\,z_j^2\ -\ c\!\int\Psi_\rho\bigl(4\theta+K_be^{-s}\bigr)z_j^2 .
\]
No $W'$ or $\chi'$ term survives, and $a$ enters only through $c$: no $\int|a|$, no $\sup|W'/W|$.
\item \emph{(signs)} Assume $\bar a<\tfrac14$, $g\ge0$, $K_b\ge0$ and $\eta$ nondecreasing. Then the first term (the \emph{inflow}) is $\ge0$, is supported on the inner ramp and is at most $c^+\!\int(\eta^2)'\chi^2WD\phi z_j^2$ with $c^+=\tfrac14+\bar a$. The second term is $\le0$: it is the \emph{damping} $-(1-4a)g\int\varepsilon\eta^2\chi^2W\phi z_j^2$ plus the $b$-absorber $-cK_b\int e^{-s}\Psi_\rho z_j^2$.
\item \emph{($b$-term)} With $C_j=(5,5,8)$, $C_{21}=1$ (all other $C_{jk}=0$) and $\Gamma_1(\Theta)\le4.000000000000001$ \tC,
\[
\Bigl|2\!\int\Psi_\rho z_j\partial_s^j(\beta z_1)\Bigr|\le\bar b\Bigl[(C_j+\tfrac{K_b}{28})\!\int e^{-s}\Psi_\rho z_j^2+\sum_kC_{jk}\!\int e^{-s}\Psi_\rho z_k^2\Bigr]+\frac{\bar b}{28}\!\int(\eta^2)'\chi^2WD\phi z_j^2+\bar b\!\int e^{-s}|(\chi^2)'|\eta^2WD\phi z_j^2 .
\]
If $c^-K_b\ge2\bar b\,(C_j+K_b/28)$, the self terms are absorbed by \emph{half} of the $b$-absorber. Then the lower-level terms obey
\[
\int_{\tau_\rho}^\infty\!\!\int e^{-s}\Psi_\rho z_k^2\le\frac{\B_k}{c^-K_b-\bar b\,(C_k+K_b/28)},
\]
where $\B_k$ is the level-$k$ total (data plus time-integrated forcing). For $K_b=10^{-3}$: $c^-K_b=2.4999995\cdot10^{-4}$ against $2\bar b\cdot8=3.5\cdot10^{-6}$, a factor $71$. The level-2 coupling is $\bar b\,\B_1/(c^-K_b-5\bar b)\le8.8\cdot10^{-4}\,\B_1$.
\item \emph{(cutoff parts of the $b$-term)} The inner-ramp part raises $c^+$ to $\tfrac14+\bar a+\bar b/28$ in the inflow. On the outer ramp $e^{-s}\le\lambda(\tau)/(190\lambda_1)$, so the outer part is time-integrable: it is at most $\frac{8\bar b\,\sup|\chi'|}{190(1-4\bar a)}\sup_\tau\int_{\rm outer}WD\phi z_j^2\le2.5\cdot10^{-8}\sup_\tau\int_{\rm outer}z_j^2$, which is added to Lemma~R's $\Delta^R_j$ (data from D5a).
\end{enumerate}
\end{lemma}
\begin{proof}
(i) $\frac{d}{d\tau}\ln\lambda=-c$ and $\frac{d}{d\tau}s_\rho=c$. Hence $\partial_\tau[\chi^2(s+\ln\lambda)W(s-s_\rho)]=-c\,\partial_s[\chi^2W]$, while $\eta$, $D$ and $\phi$ do not depend on $\tau$. Next, $2\int\Psi z_j(-cz_{j+1})=-c\int\Psi(z_j^2)'=c\int\Psi'z_j^2$ ($c$ does not depend on $s$; no boundary terms by compact support). Adding,
\[
-c\!\int\eta^2(\chi^2W)'D\phi\,z_j^2+c\!\int\bigl[(\eta^2)'\chi^2WD\phi+\eta^2(\chi^2W)'D\phi+\eta^2\chi^2W(D\phi)'\bigr]z_j^2 .
\]
The $(\chi^2W)'$ terms cancel, and $(D\phi)'/(D\phi)=D'/D+\phi'/\phi=-4\theta-K_be^{-s}$.
(ii) $c\in[c^-,c^+]$; $(\eta^2)'\ge0$; $\theta,K_b\ge0$; and $c\,4\theta D=(1-4a)ge^{-4s}$.
(iii) $\partial_s^j(\beta z_1)=\beta\sum_{i=0}^j\binom ji(-1)^iz_{j+1-i}$. The $i=0$ and $i=1$ terms together give $2\int\Psi\beta z_jz_{j+1}-2j\int\Psi\beta z_j^2=-\int\beta(\Psi'/\Psi-1+2j)\Psi z_j^2$. Write $\Psi'/\Psi=(\eta^2)'/\eta^2+(\chi^2)'/\chi^2+\Theta'/\Theta+\phi'/\phi$. The non-cutoff part satisfies $|\Theta'/\Theta+\phi'/\phi-1+2j|\le\Gamma_1+K_b/28+|2j-1|$, giving $5,5,7$; the cutoff parts give the last two terms. For $j=2$ the $i=2$ term is $2\int\Psi\beta z_2z_1$, at most $\bar b\int e^{-s}\Psi(z_2^2+z_1^2)$, which adds $1$ to $C_2$ and gives $C_{21}=1$. For the integral bound: with half the absorber, level $k$ obeys $\frac d{d\tau}\E_k+(c^-K_b-\bar b(C_k+K_b/28))\int e^{-s}\Psi z_k^2\le(\text{level-}k\text{ forcing})$; integrate.
(iv) Since $r\ge190\lambda_1$ there, $e^{-s}=\lambda/r\le\lambda(\tau)/(190\lambda_1)$, and $\int_{\tau_1}^\infty\lambda/\lambda_1\le(c^-)^{-1}$. Also $|(\chi^2)'|\le2\sup|\chi'|$, with $\sup|\chi'|=1.875/\ln2$ for the quintic ramp over one $\ln2$.
\end{proof}
\begin{remark}[why the reviewer's Gronwall route does not close, and what the residual is]
Without $\phi$, the $b$-term is a rate $\le\bar b\,C/y$ relative to $\Psi$. On the inner tail of the support, $y\in[28,56]$, this rate is $\approx\bar bC/28$ at \emph{every} time. Lemma U has no decay rate: the co-moving $W$ gives none, and the damping is only $g/y^4$. So Gronwall gives $\exp(\bar bC(\tau-\tau_\rho)/28)$, which is unbounded on the infinite horizon. The quantity $4\bar b\sup|W'/W|/28\approx1.25\cdot10^{-7}$ is the integral of that rate along a \emph{characteristic} $y(\tau)=28e^{c(\tau-\tau_\rho)}$. It is the correct bound only for a weight frozen in physical space (Prop.~B$'$), not for a ramp fixed in $y$. The factor $\phi$ supplies the missing decay $cK_be^{-s}$, exactly as $D$ supplies $c\,4\theta$. Its cost is the factor $e^{K_b/28}=1+3.6\cdot10^{-5}$ on every data conversion, of the same order as the reviewer's residual. Taking $K_b=10^{-5}$ would bring the cost to $3.6\cdot10^{-7}$, but the coupling factor in (iii) would then be $O(1)$. $K_b=10^{-3}$ is the recommended value.
\end{remark}

\section{The inner-ramp energy inequality, time-integrated}\label{sec:U}
\noindent For a centre at $y_c=\rho/\lambda(\tau_\rho)$ ($y_c=56$ for late centres, $y_c=\rho/\lambda_1\ge56$ for present ones) define, with $x=c^-(\tau-\tau_\rho)$ and $W$ increasing on $\sigma\le\frac12\ln1.5$ (so $W(\sigma(\tau,y))\le W(\ln\frac y{y_c}-x)$ on the ramp):
\begin{align*}
\Sigma_{\rm in}&:=\tfrac1{c^-}\!\int_0^\infty\sup_{y\in[28,56]}\Bigl[(\eta^2)'(y)\,D(y)\phi\,\tfrac y{28}\,W(\ln\tfrac y{y_c}-x)\Bigr]dx, &
\Sigma_{E}&:=\tfrac1{c^-}\!\int_0^\infty\sup_{y\in[28,56]}\Bigl[(\tfrac{28}y)^3D(y)\phi\,W(\ln\tfrac y{y_c}-x)\Bigr]dx,\\
S_X&:=\sup_{y\in[28,56]}W(\ln\tfrac y{y_c})\,D(y)\phi\,\tfrac y{28}, &
\Sigma_{\rm rs}^{(j)}&:=\tfrac1{c^-}\!\int_0^\infty\sup_{y\in[28,56]}\Bigl[\varepsilon(y)\bigl[\hat G_{jj}D(y)-(1-4\bar a)g\bigr]_+\eta^2\phi\,\tfrac y{28}\,W(\ln\tfrac y{y_c}-x)\Bigr]dx .
\end{align*}
These are the \emph{joint} sweeps: the time integral of the ramp supremum of $\Psi/\omega$ times the indicated profile. They replace the Phase-1 products $G_w\cdot I_W^{\rm in}\cdot\sup(y/28)\cdot\sup(\cdot)$ (§\ref{sec:num}).

\begin{proposition}[Lemma U, ramp form]\label{prop:U}
Assume: B3U and the boxes of \fn{launch_lemma.tex} \S6; the rate conditions
\[
\text{(R2)}\ \ (1-4\bar a)g\ge\hat G_{22}\Bigl(1+\tfrac g{y_R^4}\Bigr)\phi,\qquad\text{(R$<$2)}\ \ (1-4\bar a)g>\hat G_{kk}D(28)\phi\ \ (k=0,1),
\]
with $y_R=56$ (design \emph{data56}, \S\ref{sec:G}) or $y_R=28$ (Phase-1 design); the K-tables of P2 (valid for $g\in[3,6]\cdot10^5$); $c^-K_b\ge2\bar b(C_2+K_b/28)$. Then for all $\tau\ge\tau_\rho$ and $j=0,1,2$:
\[
\E_j^\rho(\tau)\ \le\ \Bigl(\sqrt{\E_j^\rho(\tau_\rho)+\mathcal I_j+\mathcal R^E_j+\mathcal R^X_j+\mathcal R^{\rm rs}_j+\mathcal F_j+\mathcal C_j+\mathcal B_j+\Delta^R_j}\ +\ \Phi_j\Bigr)^2 ,
\]
where:
\begin{itemize}[leftmargin=*,itemsep=1pt]
\item inflow $\mathcal I_j=c^+\Sigma_{\rm in}E^k_j$, with $c^+=\frac14+\bar a+\bar b/28$;
\item ramp energy terms $\mathcal R^E_j=\sum_{k\le j}K_{jk}\varepsilon_{28}\Sigma_EE^k_k$ (the $K_{jj}$ include P2's $\hat\rho\,Y_{j+1}$ interpolation remainder);
\item ramp X-terms $\mathcal R^X_j=\sum_{k>j}K_{jk}\mathcal X_k$, with
\[
\mathcal X_1=\varepsilon_{28}\Sigma_EE^k_1,\qquad
\mathcal X_2=\min\Bigl\{\varepsilon_{28}\Sigma_EE^k_2,\ \bigl(\sqrt{S_XD_0}+\sqrt{\varepsilon_{28}\Sigma_Ep^4\delta U_0}\bigr)^2\Bigr\},\qquad
\mathcal X_3=\bigl(\sqrt{S_XD_1}+\sqrt{\varepsilon_{28}\Sigma_Ep^6\delta U_0}\bigr)^2 ;
\]
\item ramp share of the rate $\mathcal R^{\rm rs}_j=\Sigma^{(j)}_{\rm rs}E^k_j$ ($=0$ for $j=0,1$ under (R$<$2); for $j=2$ the bracket is $\hat G_{22}g\,(\varepsilon-56^{-4})_+$ when $y_R=56$);
\item foot terms $\mathcal F_0=0$, $\mathcal F_1=\mathrm{footw}_1J_F$, $\mathcal F_2=\mathrm{footw}_2J_F+\mathcal S_{33}$, with $J_F=\varepsilon_{28}\Sigma_EE^k_2$ and
\[
\mathcal S_{33}=c_{33}\sqrt{2R\,D(28)\phi\,E^k_2}\Bigl(\tfrac{J_F}{\sqrt{\ln2}}+\sqrt2J_F^{3/4}J_G^{1/4}\Bigr),\quad J_G=2\bigl(\mathcal X_3+\tfrac{L_{\max}^2}4J_F\bigr),\quad c_{33}=26.92+72.30+c_{223},
\]
where $R=e^{L_{\max}\ln2}\le11.5$ and $L_{\max}=3.5213$ \tC\ (P2), and $c_{223}$ comes from Lemma~\ref{lem:C};
\item couplings $\mathcal C_j=\sum_{k<j}\hat G_{jk}\Bigl[\varepsilon_{28}\Sigma_EE^k_k+\frac{D(28)\phi\,\B_k}{(1-4\bar a)g-\hat G_{kk}D(28)\phi}\Bigr]$;
\item $b$-coupling $\mathcal B_2=\bar b\,\B_1/(c^-K_b-5\bar b)$, $\mathcal B_{0,1}=0$;
\item $\Delta^R_j$: Lemma R's outer ramp, times $D(190)\phi$, plus Lemma~\ref{lem:T}(iv);
\item $\Phi_j$: forcing of the frozen profile (Prop.~B$'$), with
\[
\Phi_j\le\frac1{c^-}\Bigl[\frac{\bar b\,p\,U_{\max}}{y_c}\Bigl(D(28)\phi\!\int_{\mathbb R}\!W(\sigma)e^{-2\sigma}d\sigma\Bigr)^{1/2}+\frac{c'_j\,U_{\max}\,e^{2a_W}\,(D(28)\phi)^{1/2}}{4\cdot28^4}\Bigl(\frac{28}{y_c}\Bigr)^2\Bigr],
\]
where $c'_j=(0.0520,0.1446,0.4324)$ (Prop.~B$'$), $U_{\max}=0.09$ and $\int We^{-2\sigma}=1.271$. For $y_c=56$, $j=2$ this is $2.0\cdot10^{-9}+3.6\cdot10^{-8}$ \tE, so $\Phi_j$ is negligible and is not in the model. (Proof: $\|\beta\partial_sU_\rho\|_\Psi\le\bar bpU_{\max}(D\phi\int We^{-2\sigma})^{1/2}e^{-s_\rho(\tau)}$; $\|\varepsilon c'_jU_\rho\|_\Psi^2\le c_j'^2U_{\max}^2D\phi\sup_s(We^{-4s})\,28^{-4}/4$ with $We^{-4\sigma}\le e^{4a_W}$, so $\sup_s We^{-4s}\le e^{4a_W}y_\rho(\tau)^{-4}$; finally $\int_{\tau_\rho}^\infty y_\rho^{-k}d\tau\le y_c^{-k}/(kc^-)$.)
\end{itemize}
Here $\B_k$ is the right-hand side at level $k$, so the levels are taken in the order $0,1,2$. Every term is a time integral or a time-uniform datum, and no rate model is used.
\end{proposition}
\begin{proof}[Proof (structure; the constants are those named)]
\emph{Energy identity.} Multiply the $z_j$-equation by $2\Psi z_j$ and integrate in $s$. The fourth-order and quasilinear part, written with the divergence-form tables of (A) (\fn{tables_exact.pkl}; $|$coef$|$, no sign cancellations) for the weight $\Psi$, splits $(\Psi)^{(m)}=\eta^2(\chi^2\Theta\phi)^{(m)}+{\rm rem}_m\cdot\Theta\phi\chi^2$. This gives:
\begin{itemize}[itemsep=0pt]
\item the global quadratic terms. After the $L^4$ identity, the $\{2,3,3\}$ IBP, the box sups and the interpolation (weight $\Psi$, constants $\Gamma_m,\tilde\kappa,\tilde\kappa_2$ of $\Theta\phi$), these are $\le\sum_{k\le j}\hat G_{jk}\int\varepsilon\Psi z_k^2-(\text{dissipation used})$. The cutoff remainder of the interpolation is P2's $\hat\rho\,Y_{j+1}$ term, which is in $K_{jj}$;
\item the cutoff commutators $\sum_{\rm pairs}|{\rm coef}|A\,{\rm rem}_mz_az_b$ on the ramp: P2's $K_{jk}$ (column $k$: pair $(k,k)$ with weight $1$, mixed pairs with $\frac12$). These include the top pair $(j+1,j+1)$; see Remark~\ref{rem:top};
\item the cutoff$\times$jet classes. Those reducible by $|z_0|\le M_0^z$, $|z_1|\le M_1$ are folded into $K$ (P2, including $K_{13}$). $(2,2,2)$ goes into $c_{33}$ (weight $72.30$ \tC) and $(2,2,3)$ into $c_{223}$ (Lemma~\ref{lem:C});
\item the ramp $L^4$ foot terms (L4\_general\_weight.md (c), weights $\mathrm{footw}_{1,2}$ \tC, P2) and the exact foot term $\frac13\int(\varepsilon\Theta\eta\eta')'|z_2|^3$, which together with $(2,2,2)$ and $(2,2,3)$ forms $\mathcal S_{33}$.
\end{itemize}
Transport and $\partial_\tau\Psi$ are handled by Lemma~\ref{lem:T}. The forcing $\hat R_\rho$ gives $\frac d{d\tau}\sqrt{\cdot}\le\|\hat R_\rho\|_\Psi$.

\emph{Rates.} By Lemma~\ref{lem:T}(ii), the net rate at level $j$ is $\int\varepsilon\eta^2\chi^2W\phi[\hat G_{jj}D-(1-4a)g]z_j^2$. By (R2) the bracket is $\le0$ on $y\ge y_R$ (it is increasing in $\varepsilon$, and zero at $y_R$ by the choice of $g$). By (R$<$2) it is $\le0$ everywhere for $j=0,1$. On $[28,y_R]$ the positive part is data, $\mathcal R^{\rm rs}_j$. The cross rates $\hat G_{jk}$, $k<j$, are bounded on the ramp by (A)'s energies, and on the bulk by the time integral $\int\!\!\int\varepsilon\Psi z_k^2\le D(28)\phi\,\B_k/s_k$, where $s_k=(1-4\bar a)g-\hat G_{kk}D(28)\phi>0$ is the damping surplus at level $k$ (dividing by $D\phi\le D(28)\phi$ converts $\eta^2\chi^2W\phi$ to $\Psi$).

\emph{Conversion to (A)'s data.} Fix $\tau$. On the ramp $\chi=1$ and $\Psi\le\eta^2W(\sigma)D\phi$, so $\int_{\rm ramp}F(y)\Psi z_k^2\le\sup_y[F\,W D\phi\,y/28]\int\omega z_k^2$. Use Minkowski for $v_\rho=v_A+\delta U$, then integrate in $\tau$ with $W(\sigma(\tau,y))\le W(\ln(y/y_c)-x)$. This gives $\Sigma_{\rm in},\Sigma_E,\Sigma_{\rm rs}$ against the time-uniform $E^k$. For the $z_{k+2}$-data ($k+2=j+1$ or $j+2$), the dissipation is only bounded in time integral. Bound $\Psi/\omega$ by its sup over $\tau\ge\tau_\rho$ and over the ramp, which is $S_X$ (attained at $\tau_\rho$, by monotonicity), and use $D_k$. The $z_2$-data at level~1 can also use (A)'s level-2 \emph{energy}; the minimum of the two bounds is valid. Finally, $\mathcal F,\mathcal S_{33}$: the sup-free form of \fn{lemmaS.tex} l.319--332 with $g=(\varepsilon\Theta)^{1/2}z_2$ on the whole ramp, using:
\begin{itemize}[itemsep=0pt]
\item $\varepsilon\Theta|z_2|^3\le(\varepsilon\Theta)_{\min}^{-1/2}|g|g^2$;
\item $F(\tau)/(\varepsilon\Theta)_{\min}\le2R\,D(28)\phi\,E^k_2$, since $\sup(\varepsilon\Theta)/\inf(\varepsilon\Theta)\le R$ over the ramp at fixed $\tau$ (P2) and $1/\omega\le2$;
\item $g'=(\varepsilon\Theta)^{1/2}(z_3+\frac L2z_2)$, so $J_G\le2(\iint\varepsilon\Theta z_3^2+\frac{L_{\max}^2}4J_F)$;
\item Hölder in time, as in (A).
\end{itemize}
\emph{Summation.} Integrate the differential inequality from $\tau_\rho$; all rates are $\le0$ after the steps above, so no Gronwall factor appears.
\end{proof}

\begin{lemma}[dissipation export, honest X-factor]\label{lem:D}
(i) Suppose (A)'s level-$k$ inequality reads $E_k'+\gamma_k'E_k+\kappa_ka_{\min}\int\varepsilon\omega(z^A_{k+2})^2\le f_k$ on $[\tau_1,\infty)$ (the $\eta_A$-cut is $1$ on $[28,350]\supset[28,56]$). Then, for every $\tau_\rho\ge\tau_1$,
\[
\int_{\tau_\rho}^\infty\!\!\int_{28}^{56}\varepsilon\omega(z^A_{k+2})^2\ \le\ D_k:=\frac{E_k(\tau_1)+\int_{\tau_1}^\infty f_k^+}{\kappa_ka_{\min}} .
\]
(ii) In the certified (A) (\fn{lemmaS_A_ibp/rigor/assembly_arb.json}), $E^A_k=H_{2,k}+(\text{inflow})+(\text{ramp\_comm})+(\text{forcing})+(\text{extra\_outer})+(\text{coupling})$. The first four forcing terms are \emph{undiscounted} time integrals of $J$-type impulses; the coupling term ($2.4\cdot10^{-7}$ of $0.048$ at level 2) is in sup form. Hence $D_k\le c_DE^A_k/(\kappa_ka_{\min})$ with $c_D=1+O(10^{-5})$, once agent A confirms this split for the repaired (A). The model's default $c_D=2$ (the reviewer's $2E$) is conservative.
(iii) \emph{Honest factor.} The t1/Phase-1 rate model uses $X_{k+2}=\gamma_k'E^A_k/(2a_{\min})$ per unit $\tau$, swept by $2I_W^{\rm in}$. Against it, the honest X-term is larger by
\[
f_X=\frac{\sup_{\rm ramp}(W\,y/28)\;D_k}{2I_W^{\rm in}\,\gamma_k'E^A_k/(2a_{\min})}=\frac{1.4713\,c_D}{\kappa_k\,I_W^{\rm in}(56)\,\gamma'_k}=\frac{4.14\,c_D}{\kappa_k}\quad\text{\tE},
\]
before the pointwise $D$ (which multiplies $S_X$ by $D(56)\phi=1.036$, against the model's $G_w=1.82$). Values: $\kappa=1.25$: $\times6.6$ ($c_D=2$), $\times3.3$ ($c_D=1$). (A) as it stands ($\kappa=0.3,0.2$ at levels $0,1$): $\times27.6/41.4$ ($c_D=2$), $\times13.8/20.7$ ($c_D=1$).
\end{lemma}
\begin{proof}
(i) Integrate on $[\tau_\rho,T]$ and drop $-E_k(T)-\gamma'\int E_k\le0$. This gives $\kappa_ka_{\min}\iint\le E_k(\tau_\rho)+\int_{\tau_\rho}^\infty f_k^+$. Also $E_k(\tau_\rho)\le E_k(\tau_1)+\int_{\tau_1}^{\tau_\rho}f_k^+$, from $E_k'\le f_k$. (ii) Read off the assembly rows. (iii) Arithmetic: $\sup_{[28,56]}W(\ln\frac y{56})\frac y{28}=1.4713$ and $I_W^{\rm in}(56)=1.4431$, both \tE\ quadrature.
\end{proof}

\begin{remark}[the top pair $(j+1,j+1)$ in Lemma U]\label{rem:top}
In Lemma U the top pair is \emph{not} dropped. In P2 (\fn{ramp_K_cert.py}, \fn{coef_arrays}, $k=j+1$, $w=1$) it is part of $K_{j,j+1}$ with data $\mathcal X_{j+1}$, so the gap found in (A) (\fn{lemmaS_F2.py:81}) does not recur. What it shares with (A) is the data: $z_{j+1}$ on the ramp is (A)'s \emph{dissipation}, which is honest only as a time integral (Lemma~\ref{lem:D}). Level 0 needs $z_1$, which is (A)'s energy; level 1 needs $z_2$, which is (A)'s level-2 energy. Only level 2 needs $z_3$, with data $D_1$. Two refinements reduce that, at the cost of new cubic classes (Lemma~\ref{lem:Y}).
\end{remark}

\begin{lemma}[column $k=j+1$ by integration by parts; IBP + Young for the top pair]\label{lem:Y}
Let $c(s)$ be the (signed, solution-dependent) cutoff coefficient of a pair, per $\varepsilon\Theta$.
\begin{enumerate}[label=(\alph*),leftmargin=*]
\item \emph{Mixed pairs} reduce exactly to energies:
  \begin{itemize}[itemsep=0pt]
  \item $\int c\varepsilon\Theta z_jz_{j+1}=-\frac12\int(c\varepsilon\Theta)'z_j^2$;
  \item $\int c\varepsilon\Theta z_{j-1}z_{j+1}=-\int(c\varepsilon\Theta)'z_{j-1}z_j-\int c\varepsilon\Theta z_j^2$;
  \item $\int c\varepsilon\Theta z_{j-2}z_{j+1}=-\int(c\varepsilon\Theta)'z_{j-2}z_j-\int c\varepsilon\Theta z_{j-1}z_j$.
  \end{itemize}
\item \emph{Top pair} (one IBP, then Young against Lemma U's own ramp dissipation):
\[
\int c\varepsilon\Theta z_{j+1}^2=-\int c\varepsilon\Theta z_jz_{j+2}+\tfrac12\int(c\varepsilon\Theta)''z_j^2\le\delta a_{\min}\!\int\varepsilon\Theta\eta^2z_{j+2}^2+\int\Bigl[\frac{c^2}{4\delta a_{\min}\eta^2}+\frac{|(c\varepsilon\Theta)''|}{2\varepsilon\Theta}\Bigr]\varepsilon\Theta z_j^2 .
\]
The first term takes $\delta a_{\min}$ from $B_c$: $\hat G_{22}$ grows by at most $1.5/(1.5-\delta)$. With $\eta=S(t)$, $c^2/\eta^2$ is bounded: near the foot $c\sim(\eta^2)'$ or $(\eta^2)''$, so $c^2/\eta^2\lesssim\eta'^2+\eta'^4/\eta^2=O(t^4)$.
\end{enumerate}
The derivatives fall on $A_i^{(r)}$: their jet-free parts are $A_i^{(r+u)}$ ($u\le2$, box-bounded); their jet parts add cubic classes with one more $z$, of the types $(2,2,k)$ and $(1,2,k)$, which go through the ramp $L^4$ bound or the S33 form. These new classes are \emph{not} bounded here. Size \tE\ (\fn{top_pair_est.py}, majorant of $|{\rm rem}_m|$, jet-free coefficients):
\begin{itemize}[itemsep=0pt]
\item at level 2 the top pair is $f_{\rm top}=0.216$ of column 3; the mixed $(2,3)$ pair is $0.73$ of it;
\item for the late centre, the Young-route data is $5.2\cdot10^{-3}$ ($\delta=0.25$), against $0.046$ for the X-route on the top pair alone;
\item the mixed pairs by (a) cost $1.5\cdot10^{-4}$.
\end{itemize}
In the model this moves the late M1 from $\times2.40$ to $\times2.65$ (§\ref{sec:num}).
\end{lemma}

\begin{lemma}[$(2,2,3)$ in sup-free S33 form]\label{lem:C}
The class $(2,2,3)$ is exactly $\int c\,\varepsilon\Theta z_2^2z_3$, with $c$ the sum over pairs $(2,2)$ with jet $z_3$ and $(2,3)$ with jet $z_2$ of $|$coef$|$, ${\rm rem}_m$ and the jet coefficient $v_X$ (P2, raw weight $\sup|c|\le7.3283$ \tC). Since $z_2^2z_3=\frac13(z_2^3)'$,
\[
\int c\,\varepsilon\Theta z_2^2z_3=-\tfrac13\int(c\varepsilon\Theta)'z_2^3,\qquad\Bigl|\frac{(c\varepsilon\Theta)'}{\varepsilon\Theta}\Bigr|\le|c'_{\rm nj}|+L_{\max}|c|+|c_J|\,|z_2| .
\]
Here $c'_{\rm nj}$ is the derivative without new jets, and $c_J$ comes from differentiating the box-bounded factor of $v_X$ ($V_1\to V_2=U_2+z_2$). So $c_{223}=\frac13(\sup|c'_{\rm nj}|+L_{\max}\sup|c|)$ enters $c_{33}$, and the $c_J$ part is a quartic ramp term $\le\sup|c_J|\int\varepsilon\Theta\eta^0z_2^4$, bounded by L4\_general\_weight.md (c) with the cutoff-free datum $J_F$. \tE: $\sup|c'_{\rm nj}|\le(17.23+4)\sup|c|$ (largest ratio $\sup|(\eta^2)^{(k+1)}|/\sup|(\eta^2)^{(k)}|$, plus $\Gamma_1$), so $c_{223}\le60.5$. The Phase-1 placeholder was $32.6$. The whole $\mathcal S_{33}$ is $\le2\cdot10^{-4}$ in the model.
\end{lemma}
