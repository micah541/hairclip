\begin{align*}
&\text{(GN1)}\quad \textstyle\int\eta^2h'^2\le K S_0^2+S_0L_\eta N_2,\quad K=\tfrac12\|(\eta^2)^{\prime\prime}\|_{L^1},\ L_\eta^2=\int\eta^2\le |I|+2\delta,\\

\end{align*}
